    89→function relative(iso: string | null | undefined): string {
    90→  if (!iso) return 'Never'
    91→  const d = new Date(iso)
    92→  if (Number.isNaN(d.getTime())) return 'Never'
    93→  return formatDistanceToNowStrict(d, { addSuffix: true })
    94→}

    96→function readVersion(obj: unknown): number | null {
    97→  if (!obj || typeof obj !== 'object') return null
    98→  const v = (obj as Record<string, unknown>).version
    99→  return typeof v === 'number' && Number.isFinite(v) ? v : null
   100→}

   334→function ConflictItem({ op }: { op: SyncOperation }) {
   335→  const [busy, setBusy] = useState(false)
   336→
   337→  const localVersion = readVersion(op.payload) ?? op.base_version
   338→  const serverVersion = readVersion(op.server_record)
   339→  const diffHint =
   340→    serverVersion === null
   341→      ? `Local v${localVersion} · Server version unknown`
   342→      : `Local v${localVersion} · Server v${serverVersion}`
   343→
   344→  async function handleResolve(choice: ConflictChoice) {
   345→    setBusy(true)
   346→    try {
   347→      await resolveConflict(op.id, choice)
   348→      toast.success(
   349→        choice === 'mine'
   350→          ? 'Conflict resolved — your version will be re-pushed'
   351→          : choice === 'theirs'
   352→            ? "Conflict resolved — server's version adopted"
   353→            : 'Conflict resolved — record marked as deleted',
   354→      )
   355→    } catch (err) {
   356→      toast.error(`Could not resolve conflict: ${(err as Error).message}`)
   357→    } finally {
   358→      setBusy(false)
   359→    }
   360→  }
   361→
   362→  return (
   363→    <div className="flex flex-col gap-3 rounded-lg border border-orange-200 bg-orange-50/50 p-3 dark:border-orange-900 dark:bg-orange-950/30 sm:flex-row sm:items-center sm:justify-between">
   364→      <div className="min-w-0">
   365→        <p className="text-sm font-medium">
   366→          {entityLabel(op.entity)}{' '}
   367→          <span className="font-mono text-xs font-normal text-muted-foreground">{op.entity_id.slice(0, 8)}</span>
   368→        </p>
   369→        <p className="mt-0.5 text-xs text-muted-foreground tabular-nums">{diffHint}</p>
   370→        <p className="mt-0.5 text-xs text-orange-700 dark:text-orange-300">
   371→          {op.last_error ?? 'A newer server version exists for this record.'}
   372→        </p>
   373→      </div>
   374→      <AlertDialog>
   375→        <AlertDialogTrigger asChild>
   376→          <Button size="sm" variant="outline" className="shrink-0" disabled={busy}>
   377→            {busy ? <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden="true" /> : null}
   378→            Resolve
   379→          </Button>
   380→        </AlertDialogTrigger>
   381→        <AlertDialogContent>
   382→          <AlertDialogHeader>
   383→            <AlertDialogTitle>Resolve sync conflict</AlertDialogTitle>
   384→            <AlertDialogDescription>
   385→              {entityLabel(op.entity)} <span className="font-mono">{op.entity_id.slice(0, 8)}</span> — {diffHint}.
   386→              Your local change conflicts with the server&apos;s newer version. Choose which one to keep; this
   387→              cannot be undone. &quot;Delete&quot; marks the record as deleted on every synced device.
   388→            </AlertDialogDescription>
   389→          </AlertDialogHeader>
   390→          <AlertDialogFooter className="flex-wrap gap-2 sm:justify-between">
   391→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   392→            <div className="flex flex-wrap gap-2">
   393→              <AlertDialogAction
   394→                className="bg-destructive text-white hover:bg-destructive/90"
   395→                onClick={() => void handleResolve('delete')}
   396→              >
   397→                Delete
   398→              </AlertDialogAction>
   399→              <AlertDialogAction
   400→                className="border bg-transparent shadow-xs hover:bg-accent hover:text-accent-foreground"
   401→                onClick={() => void handleResolve('theirs')}
   402→              >
   403→                Keep server&apos;s
   404→              </AlertDialogAction>
   405→              <AlertDialogAction onClick={() => void handleResolve('mine')}>Keep mine</AlertDialogAction>
   406→            </div>
   407→          </AlertDialogFooter>
   408→        </AlertDialogContent>
   409→      </AlertDialog>
   410→    </div>
   411→  )
   412→}

   414→function FailedOpItem({ op }: { op: SyncOperation }) {
   415→  const [busy, setBusy] = useState(false)
   416→
   417→  async function handleRetry() {
   418→    setBusy(true)
   419→    try {
   420→      await retryFailedOp(op.id)
   421→      toast.success('Operation re-queued for sync')
   422→    } catch (err) {
   423→      toast.error(`Could not retry: ${(err as Error).message}`)
   424→    } finally {
   425→      setBusy(false)
   426→    }
   427→  }
   428→
   429→  async function handleDiscard() {
   430→    setBusy(true)
   431→    try {
   432→      await discardFailedOp(op.id)
   433→      toast.success('Operation discarded — the local record is unchanged')
   434→    } catch (err) {
   435→      toast.error(`Could not discard: ${(err as Error).message}`)
   436→    } finally {
   437→      setBusy(false)
   438→    }
   439→  }
   440→
   441→  return (
   442→    <div className="flex flex-col gap-2 rounded-lg border p-3 sm:flex-row sm:items-center sm:justify-between sm:gap-4">
   443→      <div className="min-w-0 flex-1">
   444→        <p className="text-sm font-medium">
   445→          {entityLabel(op.entity)}{' '}
   446→          <span className="font-mono text-xs font-normal text-muted-foreground">{op.entity_id.slice(0, 8)}</span>
   447→          <span className="ml-2 text-xs font-normal text-muted-foreground">

   493→function SyncTab() {

   498→
   499→  const ops = useLiveQuery(async () => {
   500→    if (!ws?.id) return []
   501→    const rows = await getDb().sync_operations.where('workspace_id').equals(ws.id).toArray()
   502→    return rows.sort((a, b) => b.created_at.localeCompare(a.created_at))
   503→  }, [ws?.id])
   504→
   505→  const conflicts = useMemo(() => (ops ?? []).filter((o) => o.status === 'conflict'), [ops])
   506→  const failed = useMemo(() => (ops ?? []).filter((o) => o.status === 'failed'), [ops])
   507→  const cloudLinked = Boolean(user && ws?.cloud_linked_at)

   520→        </CardHeader>
   521→        <CardContent className="space-y-4">
   522→          <dl className="grid gap-2.5 sm:grid-cols-3">

   667→type BackupRow = Record<string, unknown>

   669→function fileWins(local: BackupRow | undefined, file: BackupRow): boolean {
   670→  // Merge policy (docs/39 §4.2): keep the newer version per record; never auto-delete.
   671→  if (!local) return true // absent locally → insert as-is
   672→  const lv = readVersion(local)
   673→  const fv = readVersion(file)
   674→  if (lv === null || fv === null) return true // no version metadata → insert/refresh
   675→  if (fv > lv) return true
   676→  if (fv < lv) return false
   677→  const lu = typeof local.updated_at === 'string' ? local.updated_at : ''
   678→  const fu = typeof file.updated_at === 'string' ? file.updated_at : ''
   679→  return fu > lu // equal versions → newer updated_at wins; tie keeps local
   680→}

   682→async function mergeBackupIntoDb(backup: ParsedBackup): Promise<{ inserted: number; updated: number; skippedNewer: number }> {
   683→  const db = getDb()
   684→  const known = new Map<string, DexieTable<BackupRow, string>>()
   685→  for (const t of db.tables) known.set(t.name, t as unknown as DexieTable<BackupRow, string>)
   686→
   687→  let inserted = 0
   688→  let updated = 0
   689→  let skippedNewer = 0
   690→
   691→  for (const [name, rows] of Object.entries(backup.tables)) {
   692→    const table = known.get(name)
   693→    if (!table) continue // unknown tables are skipped (forward compatibility)
   694→    const kp = table.schema.primKey.keyPath ?? 'id'
   695→    const keyPath = typeof kp === 'string' ? kp : kp[0]
   696→    const isSequences = name === 'document_sequences'
   697→
   698→    for (const raw of rows) {
   699→      if (!raw || typeof raw !== 'object' || Array.isArray(raw)) continue
   700→      const row = raw as BackupRow
   701→      const key = row[keyPath]
   702→      if (key === undefined || key === null || key === '') continue
   703→      if (name === 'app_settings' && row.key === 'device_id') continue // current device identity always wins
   704→
   705→      const local: BackupRow | undefined = await table.get(String(key))
   706→      if (isSequences && local) {
   707→        // Sequences never decrement (CANON §6): next_seq becomes max(local, file).
   708→        const next = Math.max(Number(local.next_seq) || 0, Number(row.next_seq) || 0)
   709→        await table.put({ ...row, next_seq: next })
   710→        updated += 1
   711→        continue
   712→      }
   713→      if (local && !fileWins(local, row)) {
   714→        skippedNewer += 1
   715→        continue
   716→      }
   717→      await table.put(row)
   718→      if (local) updated += 1
   719→      else inserted += 1
   720→    }
   721→  }
   722→
   723→  return { inserted, updated, skippedNewer }
   724→}

   773→  async function onFileChosen(file: File) {
   774→    try {
   775→      const text = await file.text()
   776→      const parsed: unknown = JSON.parse(text)
   777→      if (!parsed || typeof parsed !== 'object' || Array.isArray(parsed)) {
   778→        throw new Error('The file is not a valid JSON backup')
   779→      }
   780→      const obj = parsed as Record<string, unknown>
   781→      const meta = (obj.meta ?? {}) as Record<string, unknown>
   782→      if (meta.app !== 'invoiceflow') {
   783→        throw new Error('This file is not an InvoiceFlow backup (meta.app mismatch)')
   784→      }
   785→      const version = typeof meta.version === 'number' ? meta.version : BACKUP_FORMAT_VERSION
   786→      if (version > BACKUP_FORMAT_VERSION) {
   787→        throw new Error('This backup was created by a newer InvoiceFlow version — update the app first')
   788→      }
   789→      const rawTables = (obj.tables ?? {}) as Record<string, unknown>
   790→      const tables: Record<string, unknown[]> = {}
   791→      for (const [name, rows] of Object.entries(rawTables)) {
   792→        if (Array.isArray(rows)) tables[name] = rows
   793→      }
   794→      if (Object.keys(tables).length === 0) throw new Error('The backup contains no tables')
   795→      setPendingImport({ version, exportedAt: typeof meta.exported_at === 'string' ? meta.exported_at : null, tables })
   796→    } catch (err) {
   797→      toast.error(`Import failed: ${(err as Error).message}`)
   798→    }
