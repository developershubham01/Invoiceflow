    29→    29→export function resolveTemplate(model: UnifiedDocumentModel): DocTemplate {
    30→    30→  const id = model.kind === 'INVOICE' ? model.company.invoiceTemplate : model.company.quotationTemplate
    31→    31→  return getDocTemplate(id)
    32→    32→}
    33→    33→
    34→    34→export function renderDocumentPdf(model: UnifiedDocumentModel): jsPDF {
    35→    35→  const tpl = resolveTemplate(model)
    36→    36→  const doc = new jsPDF({ unit: 'mm', format: 'a4', compress: true })
    37→    37→  const pageW = doc.internal.pageSize.getWidth()
    38→    38→  const pageH = doc.internal.pageSize.getHeight()
    39→    39→  const ctx: Ctx = {
    40→    40→    doc,
    41→    41→    model,
    42→    42→    tpl,
    43→    43→    th: tpl.theme,
    44→    44→    pageW,
    45→    45→    pageH,
    46→    46→    ml: tpl.layout === 'sidebar' ? 22 : 14,
    47→    47→    mr: 14,
    48→    48→  }
    49→    49→
    50→    50→  let y = drawHeader(ctx)
    51→    51→
    52→    52→  y = drawBillTo(ctx, y)
    53→    53→  y = drawItems(ctx, y)
    54→    54→  y = drawTotals(ctx, y)
    55→    55→  y = drawWordsBankNotes(ctx, y)
    56→    56→  drawTermsSignature(ctx, y)
    57→    57→  drawDecorations(ctx)
    58→    58→
    59→    59→  return doc
    60→    60→}

    64→    64→function companyLines(ctx: Ctx): string[] {
    65→    65→  const c = ctx.model.company
    66→    66→  return [
    67→    67→    ...c.addressLines.slice(0, 2),
    68→    68→    c.gstin ? `GSTIN: ${c.gstin}` : null,
    69→    69→    c.pan ? `PAN: ${c.pan}` : null,
    70→    70→    c.phone ? `Phone: ${c.phone}` : null,
    71→    71→    c.email ? `Email: ${c.email}` : null,
    72→    72→  ].filter((x): x is string => Boolean(x))
    73→    73→}

    75→    75→function dateLines(ctx: Ctx): string[] {
    76→    76→  const m = ctx.model
    77→    77→  const fmt = (d: string) => formatDateDisplay(d, m.company.dateFormat)
    78→    78→  const out = [`Date: ${fmt(m.date)}`]
    79→    79→  if (m.dueDate) out.push(`Due: ${fmt(m.dueDate)}`)
    80→    80→  if (m.validUntil) out.push(`Valid until: ${fmt(m.validUntil)}`)
    81→    81→  return out
    82→    82→}

    88→    88→function drawLogo(ctx: Ctx, x: number, y: number, size = 14): boolean {
    89→    89→  if (!ctx.model.company.logoData) return false
    90→    90→  try {
    91→    91→    const fmt = ctx.model.company.logoData.includes('image/png') ? 'PNG' : 'JPEG'
    92→    92→    ctx.doc.addImage(ctx.model.company.logoData, fmt, x, y, size, size)
    93→    93→    return true
    94→    94→  } catch {
    95→    95→    return false // invalid image data — skip gracefully
    96→    96→  }
    97→    97→}

    99→    99→function initials(ctx: Ctx): string {
   100→   100→  return ctx.model.company.name
   101→   101→    .split(/\s+/)
   102→   102→    .filter(Boolean)
   103→   103→    .slice(0, 2)

   105→   105→    .join('') || 'IF'

   110→   110→function drawHeader(ctx: Ctx): number {
   111→   111→  switch (ctx.tpl.layout) {
   112→   112→    case 'band': return drawBandHeader(ctx)
   113→   113→    case 'block': return drawBlockHeader(ctx)
   114→   114→    case 'sidebar': return drawSidebarHeader(ctx)
   115→   115→    case 'centered': return drawCenteredHeader(ctx)
   116→   116→    case 'corner': return drawCornerHeader(ctx)
   117→   117→    case 'split': return drawSplitHeader(ctx)
   118→   118→    case 'minimal': return drawMinimalHeader(ctx)
   119→   119→    case 'stack': return drawStackHeader(ctx)
   120→   120→  }
   121→   121→}

   124→   124→function drawBandHeader(ctx: Ctx): number {
   125→   125→  const { doc, th, ml, pageW } = ctx
   126→   126→  const base = ctx.tpl.serif ? 'times' : 'helvetica'
   127→   127→  const y = 16
   128→   128→  doc.setFillColor(...th.accent)
   129→   129→  doc.rect(0, 0, pageW, 4, 'F')
   130→   130→
   131→   131→  const hasLogo = drawLogo(ctx, ml, y)
   132→   132→  const textX = ml + (hasLogo ? 18 : 0)
   133→   133→  doc.setFont(base, 'bold')
   134→   134→  doc.setFontSize(16)
   135→   135→  doc.setTextColor(...th.accentDark)
   136→   136→  doc.text(doc.splitTextToSize(ctx.model.company.name, 90)[0], textX, y + 5)
   137→   137→  doc.setFont('helvetica', 'normal')
   138→   138→  doc.setFontSize(8)
   139→   139→  doc.setTextColor(...th.muted)
   140→   140→  companyLines(ctx).slice(0, 4).forEach((line, i) => doc.text(line, textX, y + 9 + i * 3.6))
   141→   141→
   142→   142→  doc.setFont(base, 'bold')
   143→   143→  doc.setFontSize(15)
   144→   144→  doc.setTextColor(...th.accent)
   145→   145→  doc.text(docTitle(ctx), pageW - ctx.mr, y + 5, { align: 'right' })
   146→   146→  doc.setFontSize(8.5)
   147→   147→  doc.setTextColor(...DARK)
   148→   148→  doc.text(`No. ${ctx.model.number}`, pageW - ctx.mr, y + 10.5, { align: 'right' })
   149→   149→  doc.setFont('helvetica', 'normal')
   150→   150→  doc.setTextColor(...th.muted)
   151→   151→  dateLines(ctx).forEach((line, i) => doc.text(line, pageW - ctx.mr, y + 14.5 + i * 3.5, { align: 'right' }))
   152→   152→
   153→   153→  return y + (hasLogo ? 22 : 24)
   154→   154→}

   225→   225→function drawCenteredHeader(ctx: Ctx): number {
   226→   226→  const { doc, th, ml, pageW } = ctx
   227→   227→  const base = ctx.tpl.serif ? 'times' : 'helvetica'
   228→   228→  const cx = pageW / 2
   229→   229→  const cw = pageW - ml - ctx.mr
   230→   230→  let y = 16
   231→   231→
   232→   232→  doc.setFont(base, 'bold')
   233→   233→  doc.setFontSize(17)
   234→   234→  doc.setTextColor(...th.accentDark)
   235→   235→  doc.text(doc.splitTextToSize(ctx.model.company.name, cw)[0], cx, y + 5, { align: 'center' })
   236→   236→  doc.setFont('helvetica', 'normal')
   237→   237→  doc.setFontSize(7.8)
   238→   238→  doc.setTextColor(...th.muted)
   239→   239→  const contact = companyLines(ctx).join('  ·  ')
   240→   240→  const contactLines = doc.splitTextToSize(contact, cw)
   241→   241→  doc.text(contactLines, cx, y + 9.5, { align: 'center' })
   242→   242→  y += 9.5 + contactLines.length * 3.4 + 3
   243→   243→
   244→   244→  doc.setFont(base, 'bold')
   245→   245→  doc.setFontSize(13.5)
   246→   246→  doc.setTextColor(...th.accent)
   247→   247→  doc.text(docTitle(ctx), cx, y + 3, { align: 'center' })
   248→   248→  doc.setFont('helvetica', 'normal')
   249→   249→  doc.setFontSize(8.5)
   250→   250→  doc.setTextColor(...DARK)
   251→   251→  doc.text(`No. ${ctx.model.number}`, cx, y + 7.5, { align: 'center' })
   252→   252→  doc.setTextColor(...th.muted)
   253→   253→  doc.setFontSize(8)
   254→   254→  const dateLine = dateLines(ctx).join('   ·   ')
   255→   255→  doc.text(dateLine, cx, y + 11.2, { align: 'center' })
   256→   256→  y += 14.5
   257→   257→
   258→   258→  doc.setDrawColor(...th.accent)
   259→   259→  doc.setLineWidth(0.7)
   260→   260→  doc.line(ml, y, pageW - ctx.mr, y)

   263→   263→  doc.line(ml, y + 1.4, pageW - ctx.mr, y + 1.4)
   264→   264→  doc.setLineWidth(0.2)
   265→   265→  return y + 7
   266→   266→}

   311→   311→function drawSplitHeader(ctx: Ctx): number {
   312→   312→  const { doc, th, ml, pageW } = ctx
   313→   313→  const base = ctx.tpl.serif ? 'times' : 'helvetica'
   314→   314→  const blockW = 64
   315→   315→  const blockH = 30
   316→   316→
   317→   317→  doc.setFillColor(...th.accent)
   318→   318→  doc.rect(0, 0, blockW, blockH, 'F')
   319→   319→  const hasLogo = drawLogo(ctx, ml, 6, 11)
   320→   320→  if (hasLogo) {
   321→   321→    doc.setFont(base, 'bold')
   322→   322→    doc.setFontSize(11)
   323→   323→    doc.setTextColor(...WHITE)
   324→   324→    const nameLines = doc.splitTextToSize(ctx.model.company.name, blockW - ml - 8)
   325→   325→    doc.text(nameLines[0] ?? '', ml + 14, 12)
   326→   326→    if (nameLines[1]) {
   327→   327→      doc.setFontSize(8.5)
   328→   328→      doc.text(nameLines[1], ml + 14, 16)
   329→   329→    }
   330→   330→  } else {
   331→   331→    doc.setFont(base, 'bold')
   332→   332→    doc.setFontSize(13)
   333→   333→    doc.setTextColor(...WHITE)
   334→   334→    const nameLines = doc.splitTextToSize(ctx.model.company.name, blockW - ml - 6)
   335→   335→    doc.text(nameLines.slice(0, 2), ml, 10, { baseline: 'top' })
   336→   336→  }
   337→   337→  doc.setFont('helvetica', 'normal')
   338→   338→  doc.setFontSize(7.4)
   339→   339→  doc.setTextColor(255, 228, 235)
   340→   340→  const biz = ctx.model.company.gstin ? `GSTIN: ${ctx.model.company.gstin}` : ''
   341→   341→  if (biz) doc.text(biz, ml, blockH - 4)
   342→   342→
   343→   343→  const rightX = blockW + 8
   344→   344→  doc.setFont('helvetica', 'normal')
   345→   345→  doc.setFontSize(7.8)
   346→   346→  doc.setTextColor(...th.muted)
   347→   347→  companyLines(ctx)
   348→   348→    .filter((l) => !l.startsWith('GSTIN'))
   349→   349→    .slice(0, 4)
   350→   350→    .forEach((line, i) => doc.text(line, rightX, 9 + i * 3.6))
   351→   351→
   352→   352→  let y = blockH + 7
   353→   353→  doc.setFont(base, 'bold')
   354→   354→  doc.setFontSize(14)
   355→   355→  doc.setTextColor(...th.accentDark)
   356→   356→  doc.text(docTitle(ctx), ml, y)
   357→   357→  doc.setFontSize(8.5)
   358→   358→  doc.setTextColor(...DARK)
   359→   359→  doc.text(`No. ${ctx.model.number}`, pageW - ctx.mr, y, { align: 'right' })
   360→   360→  doc.setFont('helvetica', 'normal')
   361→   361→  doc.setTextColor(...th.muted)
   362→   362→  dateLines(ctx).forEach((line, i) => doc.text(line, pageW - ctx.mr, y + 4.5 + i * 3.5, { align: 'right' }))
   363→   363→  y += 4.5 + dateLines(ctx).length * 3.5 + 2
   364→   364→
   365→   365→  doc.setDrawColor(...th.line)
   366→   366→  doc.setLineWidth(0.3)
   367→   367→  doc.line(ml, y, pageW - ctx.mr, y)
   368→   368→  doc.setLineWidth(0.2)
   369→   369→  return y + 6
   370→   370→}

   487→   487→function drawItems(ctx: Ctx, y: number): number {

   492→   492→    'Taxable', intra ? 'CGST' : 'IGST', intra ? 'SGST' : '', 'Amount',
   493→   493→  ].filter((h) => h !== '')]

   496→   496→      String(it.position),
   497→   497→      it.description,

   505→   505→    if (intra) row.push(formatMoneyPlain(it.sgstPaise))
   506→   506→    row.push(formatMoneyPlain(it.totalPaise))

   509→   509→
   510→   510→  const headStyles: Record<string, unknown> = { fontStyle: 'bold', fontSize: 7.4 }
   511→   511→  let headBottom = 0

   520→   520→    headStyles.textColor = th.accentDark
   521→   521→  }
   522→   522→
   523→   523→  autoTable(doc, {
   524→   524→    head,
   525→   525→    body,
   526→   526→    startY: y,

   539→   539→      ...(intra ? { 8: { cellWidth: 19, halign: 'right' } } : {}),

   541→   541→    },
   542→   542→    didDrawCell: (data) => {

   558→   558→function drawTotals(ctx: Ctx, y: number): number {
